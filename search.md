<script src="{{ '/pagefind/pagefind-ui.js' | relative_url }}"></script>
<script>
  window.addEventListener("DOMContentLoaded", function () {
    if (typeof PagefindUI === "undefined") {
      document.getElementById("search").textContent =
        "The search index isn't available in this build (it's generated by Pagefind after the Jekyll build).";
      return;
    }
    new PagefindUI({ element: "#search", showSubResults: true, showImages: false, resetStyles: false });

    var input = document.querySelector("#search input");
    var params = new URLSearchParams(window.location.search);
    if (input && params.get("q")) {
      input.value = params.get("q");
      input.dispatchEvent(new Event("input", { bubbles: true }));
    }

    // Report searches to GA4 once the visitor pauses typing.
    var timer;
    if (input) input.addEventListener("input", function () {
      clearTimeout(timer);
      var term = input.value.trim();
      timer = setTimeout(function () {
        if (term.length >= 3 && typeof window.gtag === "function") {
          window.gtag("event", "search", { search_term: term });
        }
      }, 1500);
    });
  });
</script>
